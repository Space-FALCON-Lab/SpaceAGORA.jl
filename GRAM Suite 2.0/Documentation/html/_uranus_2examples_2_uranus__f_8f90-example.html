6 PROGRAM uranus_f
7  use, INTRINSIC :: iso_c_binding
8  USE gramstructs

11  TYPE(C_PTR) :: uranus, uranus2
12  TYPE(GramTime_F) :: time
13  TYPE(Position_F) :: pos, pos2
14  TYPE(DynamicsState_F) :: dstate, dstate2
15  CHARACTER(KIND=C_CHAR, LEN = 500) :: spicePath
16  CHARACTER(KIND=C_CHAR, LEN = 100) :: version
17  INTEGER(C_SIZE_T) :: bufferSize
18  INTEGER(C_SIZE_T) :: stringLength
19  INTEGER(C_INT) :: errorFlag;
20  TYPE(GasesState_F) :: gstate, gstate2

25 ! TYPE(WindsState_F) :: wstate, wstate2

27  CHARACTER(LEN=*), PARAMETER :: FMT = "(A, T27, ES13.6, 1X, ES13.6)"

29  write(*,*) "================="
30  write(*,*) "A FORTRAN Example"

34  ! The first call must be to initialize the Spice path.
35  call trygetspicepath_u(spicepath, 500);
36  call initialize_u(spicepath)
37 
38  ! Create a Uranus atmosphere. A pointer is returned that is our
39  ! handle on the uranus model. More than one Uranus atmosphere
40  ! can be created. (Also see deleteAtmosphere_U() below.)
41  uranus = createatmosphere_u()
42  uranus2 = createatmosphere_u()
43 
44  ! Get and print the version string.
45  ! On input, bufferSize is the length of the character array.
46  ! On output, bufferSize is the length of the version string.
47  buffersize = 100
48  stringlength = getversionstring_u(uranus, version, buffersize)
49  write(*,*) version(1:stringlength)
50  write(*,*)
51 
52  ! Set the initial seed, the relative step size, and the perturbation scale factors.
53  call setseed_u(uranus, 1001)
54  call setminrelativestepsize_u(uranus, 0.5d0)
55  call setperturbationscales_u(uranus, 1.5d0, 1.5d0, 1.5d0)
56  call setseed_u(uranus2, 1001)
57  call setminrelativestepsize_u(uranus2, 0.5d0)
58  call setperturbationscales_u(uranus2, 1.5d0, 1.5d0, 1.5d0)
59 
60  ! Set the start time.
61  time%year = 2020
62  time%month = 3
63  time%day = 15
64  time%hour = 0
65  time%minute = 0
66  time%seconds = 0.0d0
67  time%frame = 1
68  time%scale = 1
69  call setstarttime_u(uranus, time)
70  call setstarttime_u(uranus2, time)
71 
72  ! Set the position.
73  pos%height = 50.0
74  pos%latitude = 22.0
75  pos%longitude = 48.0
76  pos%elapsedTime = 100.0
77  pos%isPlanetoCentric = 1
78  call setposition_u(uranus, pos)
79  pos%height = 1000.0
80  call setposition_u(uranus2, pos)
81 
82  ! Update the model.
83  errorflag = update_u(uranus)
84  errorflag = update_u(uranus2)
85 
86  ! The position and other output is returned in the Position_F structure.
87  call getposition_u(uranus, pos)
88  call getposition_u(uranus2, pos2)
89  write(*,*) " Uranus #1 Uranus #2"
90  write(*,fmt) "Total Radius:", pos%totalRadius, pos2%totalRadius
91  write(*,fmt) "Gravity:", pos%gravity, pos2%gravity
92  write(*,*)
93 
94  ! The ephemeris values are returned in the EphemerisState_F structure
95  call getephemerisstate_u(uranus, estate)
96  call getephemerisstate_u(uranus2, estate2)
97  write(*,fmt) "Solar Time:", estate%solarTime, estate2%solarTime
98  write(*,fmt) "Longitude of the Sun:", estate%longitudeSun, estate2%longitudeSun
99  write(*,*)
100 
101  ! The atmosphere state is returned in multiple structures:
102  ! DynamicsState_F, DensityState_F, WindsState_F, GasesState_F
103  call getdynamicsstate_u(uranus, dstate)
104  call getdynamicsstate_u(uranus2, dstate2)
105  write(*,fmt) "Temperature:", dstate%temperature, dstate2%temperature
106  write(*,fmt) "Pressure:", dstate%pressure, dstate2%pressure
107  write(*,fmt) "Density:", dstate%density, dstate2%density
108  write(*,fmt) "Pressure Scale Height:", dstate%pressureScaleHeight, dstate2%pressureScaleHeight
109  write(*,fmt) "Density Scale Height:", dstate%densityScaleHeight, dstate2%densityScaleHeight
110  write(*,*)
111 
112  ! Get and print perturbed density
113  call getdensitystate_u(uranus, rstate)
114  call getdensitystate_u(uranus2, rstate2)
115  write(*,fmt) "Mean Density:", rstate%density, rstate2%density
116  write(*,fmt) "Perturbed Density:", rstate%perturbedDensity, rstate2%perturbedDensity
117  write(*,fmt) "Perturbation Percent:", rstate%densityPerturbation, rstate2%densityPerturbation
118  write(*,*)
119 
120  ! Get and print gases.
121  call getgasesstate_u(uranus, gstate, dihydrogen, methane, helium)
122  call getgasesstate_u(uranus2, gstate2, dihydrogen2, methane2, helium2)
123  write(*,fmt) "Average Molecular Weight:", gstate%averageMolecularWeight, gstate2%averageMolecularWeight
124  write(*,fmt) "Dihydrogen Mole Fraction:", dihydrogen%moleFraction, dihydrogen2%moleFraction
125  write(*,fmt) "Methane Mole Fraction:", methane%moleFraction, methane2%moleFraction
126  write(*,fmt) "Helium Mole Fraction:", helium%moleFraction, helium2%moleFraction
127  write(*,*)
128 
129  ! Get and print wind data
130 ! call getWindsState_U(uranus, wstate)
131 ! call getWindsState_U(uranus2, wstate2)
132 ! write(*,FMT) "EW Wind:", wstate%ewWind, wstate2%ewWind
133 ! write(*,FMT) "Perturbed EW Wind:", wstate%perturbedEWWind, wstate2%perturbedEWWind
134 
135  ! Memory is freed up in the deleteAtmosphere call.
136  call deleteatmosphere_u(uranus)
137  call deleteatmosphere_u(uranus2)
