207 size_t MAP::getUpperIndex(greal value, size_t size, T &array)
208 {
209  size_t index = 0;
210  if (value < array[0] || value > array[size - 1]) {
211  return index;
212  }
213  while (index < size - 1 && value >= array[index]) {
214  ++index;
215  }
216  return index;
217 }
